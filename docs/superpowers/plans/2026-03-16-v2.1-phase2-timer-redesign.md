# Phase 2: Timer Controls Redesign

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the ugly bare-button timer controls with a polished, phone-friendly design featuring large time display, icon+text pill buttons, countdown progress bar, and zero-pulse animation.

**Architecture:** Keep the existing timer logic (refs, intervals, countdown/stopwatch modes). Replace JSX layout and move to CSS Module with theme variables. Add Lucide icons (Play, Pause, RotateCcw). Add linear progress bar for countdown mode.

**Tech Stack:** TypeScript, React 19, CSS Modules, Lucide React

## Task 2: Rewrite TimerElement with new UI

**Files:**
- Modify: `engines/web-ui/src/components/elements/TimerElement.tsx`

- [ ] **Step 1: Rewrite TimerElement**

```typescript
import { useEffect, useRef, useState, useCallback } from 'react';
import { Play, Pause, RotateCcw } from 'lucide-react';
import type { ElementProps } from './registry';
import type { FormElementTimer } from '@engine/types.js';
import styles from './TimerElement.module.css';

export function TimerElement({ element, onFormChange }: ElementProps) {
  const el = element as FormElementTimer;
  const [elapsed, setElapsed] = useState(0);
  const [running, setRunning] = useState(true);
  const intervalRef = useRef<ReturnType<typeof setInterval>>(undefined);
  const startTimeRef = useRef(Date.now());
  const pausedElapsedRef = useRef(0);
  const stoppedRef = useRef(false);

  const stopTimer = useCallback(() => {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = undefined;
    }
  }, []);

  const tick = useCallback(() => {
    if (stoppedRef.current) return;
    const secs = pausedElapsedRef.current + Math.floor((Date.now() - startTimeRef.current) / 1000);
    setElapsed(secs);
    const display = el.direction === 'countdown'
      ? Math.max(0, el.durationSeconds - secs)
      : secs;
    onFormChange(el.fieldName, String(display));

    if (el.direction === 'countdown' && secs >= el.durationSeconds) {
      stoppedRef.current = true;
      setRunning(false);
    }
  }, [el.fieldName, el.durationSeconds, el.direction, onFormChange]);

  const startTimer = useCallback(() => {
    stopTimer();
    stoppedRef.current = false;
    startTimeRef.current = Date.now();
    intervalRef.current = setInterval(tick, 1000);
  }, [tick, stopTimer]);

  useEffect(() => {
    startTimer();
    return () => {
      stoppedRef.current = true;
      stopTimer();
    };
  }, [startTimer, stopTimer]);

  const handlePlayPause = () => {
    if (running) {
      pausedElapsedRef.current = elapsed;
      stoppedRef.current = true;
      stopTimer();
      setRunning(false);
    } else {
      startTimeRef.current = Date.now();
      startTimer();
      setRunning(true);
    }
  };

  const handleReset = () => {
    pausedElapsedRef.current = 0;
    setElapsed(0);
    startTimeRef.current = Date.now();
    const resetDisplay = el.direction === 'countdown' ? el.durationSeconds : 0;
    onFormChange(el.fieldName, String(resetDisplay));
    if (running) {
      startTimer();
    } else {
      stoppedRef.current = true;
      stopTimer();
    }
  };

  const display = el.direction === 'countdown'
    ? Math.max(0, el.durationSeconds - elapsed)
    : elapsed;
  const mins = Math.floor(display / 60);
  const secs = display % 60;
  const isExpired = el.direction === 'countdown' && elapsed >= el.durationSeconds;
  const progressPct = el.direction === 'countdown'
    ? Math.max(0, 100 * (1 - elapsed / el.durationSeconds))
    : 0;

  return (
    <div className={styles.timer}>
      <span className={styles.label}>{el.label}</span>
      <span className={`${styles.display}${isExpired ? ` ${styles.expired}` : ''}`}>
        {String(mins).padStart(2, '0')}:{String(secs).padStart(2, '0')}
      </span>
      {el.direction === 'countdown' && (
        <div className={styles.progressBar}>
          <div
            className={`${styles.progressFill}${isExpired ? ` ${styles.expired}` : ''}`}
            style={{ width: `${progressPct}%` }}
          />
        </div>
      )}
      <div className={styles.controls}>
        <button type="button" className={styles.controlButton} onClick={handlePlayPause}>
          {running ? <Pause size={16} /> : <Play size={16} />}
          {running ? 'Pause' : 'Play'}
        </button>
        <button type="button" className={styles.controlButton} onClick={handleReset}>
          <RotateCcw size={16} />
          Reset
        </button>
      </div>
    </div>
  );
}
```
