# Phase 4: History Screen

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the History placeholder with a reverse-chronological step completion log across all workflows, with date range and workflow filters, and a step detail popup showing input/output parameters.

**Architecture:** Two engine prerequisites first (TraceEntry timestamps, CompletedWorkflow.stepParams). Then localStorage persistence for completed workflows. Finally, the History UI with filter controls and detail popup.

**Tech Stack:** TypeScript, React 19, CSS Modules, Lucide React

private completeWorkflow(instanceId: string, finalSnap: any): void {
  const active = this._active.find(w => w.id === instanceId);
  if (!active) return;

  const unsub = this._coordinatorUnsubs.get(instanceId);
  if (unsub) {
    unsub();
    this._coordinatorUnsubs.delete(instanceId);
  }

  const completed: CompletedWorkflow = {
    id: active.id,
    sourceSpecId: active.sourceSpecId,
    name: active.name,
    localId: active.localId,
    version: active.version,
    finalState: finalSnap.workflowState,
    startedAt: active.startedAt,
    finishedAt: Date.now(),
    trace: finalSnap.trace ?? [],
    properties: finalSnap.properties ?? {},
    stepParams: finalSnap.stepParams ?? {},  // ADD THIS
  };

  this._active = this._active.filter(w => w.id !== instanceId);

  if (this._focusedActiveId === instanceId) {
    this._focusedActiveId = this._active.length > 0 ? this._active[0].id : null;
  }

  active.coordinator.reset();

  this._completed = [completed, ...this._completed];
  this.publish();
}
```

```

Add to the constructor (or as an initializer if no constructor exists):

```typescript
constructor() {
  this._completed = WorkflowManager.loadCompleted();
  this._snapshot = { loaded: [], active: [], completed: this._completed, focusedActiveId: null };
}

private static loadCompleted(): CompletedWorkflow[] {
  try {
    const raw = localStorage.getItem(WorkflowManager.STORAGE_KEY);
    if (!raw) return [];
    return JSON.parse(raw) as CompletedWorkflow[];
  } catch {
    return [];
  }
}

private persistCompleted(): void {
  try {
    // Cap workflows and trace entries
    const capped = this._completed.slice(0, WorkflowManager.MAX_COMPLETED).map(wf => ({
      ...wf,
      trace: wf.trace.slice(0, WorkflowManager.MAX_TRACE_PER_WORKFLOW),
    }));
    localStorage.setItem(WorkflowManager.STORAGE_KEY, JSON.stringify(capped));
  } catch {
    // Storage full — silently fail
  }
}
```

```typescript
import { X } from 'lucide-react';
import styles from './StepDetailPopup.module.css';

interface StepDetailPopupProps {
  stepType: string;
  description: string;
  completedAt: number;
  inputParameters: Record<string, string>;
  outputParameters: Record<string, string>;
  onClose: () => void;
}

export function StepDetailPopup({ stepType, description, completedAt, inputParameters, outputParameters, onClose }: StepDetailPopupProps) {
  const date = new Date(completedAt);
  const inputEntries = Object.entries(inputParameters);
  const outputEntries = Object.entries(outputParameters);

  return (
    <div className={styles.overlay} onClick={onClose}>
      <div className={styles.panel} onClick={(e) => e.stopPropagation()}>
        <div className={styles.header}>
          <h3 className={styles.title}>{description}</h3>
          <button className={styles.closeButton} onClick={onClose}><X size={18} /></button>
        </div>

        <div className={styles.meta}>
          <div className={styles.metaRow}>Type: {stepType}</div>
          <div className={styles.metaRow}>Completed: {date.toLocaleString()}</div>
        </div>

        <h4 className={styles.sectionTitle}>Input Parameters</h4>
        {inputEntries.length > 0 ? (
          <table className={styles.paramTable}>
            <tbody>
              {inputEntries.map(([key, value]) => (
                <tr key={key}><td>{key}</td><td>{value}</td></tr>
              ))}
            </tbody>
          </table>
        ) : (
          <p className={styles.emptyParams}>None</p>
        )}

        <h4 className={styles.sectionTitle}>Output Parameters</h4>
        {outputEntries.length > 0 ? (
          <table className={styles.paramTable}>
            <tbody>
              {outputEntries.map(([key, value]) => (
                <tr key={key}><td>{key}</td><td>{value}</td></tr>
              ))}
            </tbody>
          </table>
        ) : (
          <p className={styles.emptyParams}>None</p>
        )}
      </div>
    </div>
  );
}
```

- [ ] **Step 2: Implement HistoryScreen.tsx**

```typescript
import { useState, useMemo } from 'react';
import { useManagerSnapshot, useWorkflowManager } from '../../manager/useWorkflowManager';
import { StepDetailPopup } from '../StepDetailPopup';
import styles from './HistoryScreen.module.css';

type DateFilter = 'today' | 'week' | 'all';

interface HistoryEntry {
  workflowId: string;
  workflowName: string;
  stepOid: string;
  stepType: string;
  description: string;
  completedAt: number;
  inputParameters: Record<string, string>;
  outputParameters: Record<string, string>;
}

function formatRelativeTime(ts: number): string {
  const diff = Date.now() - ts;
  const mins = Math.floor(diff / 60000);
  if (mins < 1) return 'just now';
  if (mins < 60) return `${mins}m ago`;
  const hours = Math.floor(mins / 60);
  if (hours < 24) return `${hours}h ago`;
  return new Date(ts).toLocaleDateString();
}

export function HistoryScreen() {
  const snapshot = useManagerSnapshot();
  const [dateFilter, setDateFilter] = useState<DateFilter>('all');
  const [workflowFilter, setWorkflowFilter] = useState<string>('all');
  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);

  // Build flat list of history entries from all completed workflows
  const allEntries = useMemo(() => {
    const entries: HistoryEntry[] = [];
    const autoTypes = new Set(['START', 'END', 'PARALLEL', 'WAIT ALL', 'WAIT ANY', 'SELECT 1', 'SELECT_1', 'SCRIPT', 'MATH']);

    for (const wf of snapshot.completed) {
      for (const trace of wf.trace) {
        if (trace.state !== 'COMPLETED') continue;
        const params = wf.stepParams?.[trace.step_oid];
        if (!params) continue;
        if (autoTypes.has(params.stepType)) continue;

        entries.push({
          workflowId: wf.id,
          workflowName: wf.localId,
          stepOid: trace.step_oid,
          stepType: params.stepType,
          description: params.description || params.label || trace.step_oid,
          completedAt: trace.timestamp ?? wf.finishedAt,
          inputParameters: params.inputParameters,
          outputParameters: params.outputParameters,
        });
      }
    }

    entries.sort((a, b) => b.completedAt - a.completedAt);
    return entries;
  }, [snapshot.completed]);

  // Get unique workflow names for filter dropdown
  const workflowNames = useMemo(() => {
    const names = new Set(allEntries.map(e => e.workflowName));
    return [...names].sort();
  }, [allEntries]);

  // Apply filters
  const filteredEntries = useMemo(() => {
    let entries = allEntries;

    // Date filter
    if (dateFilter === 'today') {
      const todayStart = new Date();
      todayStart.setHours(0, 0, 0, 0);
      entries = entries.filter(e => e.completedAt >= todayStart.getTime());
    } else if (dateFilter === 'week') {
      const weekAgo = Date.now() - 7 * 24 * 60 * 60 * 1000;
      entries = entries.filter(e => e.completedAt >= weekAgo);
    }

    // Workflow filter
    if (workflowFilter !== 'all') {
      entries = entries.filter(e => e.workflowName === workflowFilter);
    }

    return entries;
  }, [allEntries, dateFilter, workflowFilter]);

  return (
    <div className={styles.screen}>
      <div className={styles.filters}>
        {(['today', 'week', 'all'] as DateFilter[]).map(f => (
          <button
            key={f}
            className={`${styles.filterPill}${dateFilter === f ? ` ${styles.active}` : ''}`}
            onClick={() => setDateFilter(f)}
          >
            {f === 'today' ? 'Today' : f === 'week' ? 'Last 7 days' : 'All'}
          </button>
        ))}
        {workflowNames.length > 1 && (
          <select
            className={styles.workflowFilter}
            value={workflowFilter}
            onChange={(e) => setWorkflowFilter(e.target.value)}
          >
            <option value="all">All Workflows</option>
            {workflowNames.map(name => (
              <option key={name} value={name}>{name}</option>
            ))}
          </select>
        )}
      </div>

      {filteredEntries.length > 0 ? (
        <ul className={styles.list}>
          {filteredEntries.map((entry, i) => (
            <li key={`${entry.stepOid}-${i}`} className={styles.item} onClick={() => setSelectedEntry(entry)}>
              <span className={styles.workflowBadge}>{entry.workflowName}</span>
              <div className={styles.stepInfo}>
                <div className={styles.stepType}>{entry.stepType}</div>
                <div className={styles.stepDesc}>{entry.description}</div>
              </div>
              <span className={styles.stepTime}>{formatRelativeTime(entry.completedAt)}</span>
            </li>
          ))}
        </ul>
      ) : (
        <p className={styles.empty}>No completed steps found</p>
      )}

      {selectedEntry && (
        <StepDetailPopup
          stepType={selectedEntry.stepType}
          description={selectedEntry.description}
          completedAt={selectedEntry.completedAt}
          inputParameters={selectedEntry.inputParameters}
          outputParameters={selectedEntry.outputParameters}
          onClose={() => setSelectedEntry(null)}
        />
      )}
    </div>
  );
}
```
