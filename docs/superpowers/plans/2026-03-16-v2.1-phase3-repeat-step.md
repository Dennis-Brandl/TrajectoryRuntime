# Phase 3: REPEAT Step Support

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add user-driven "jump back to a previous step" capability. The user picks one or more completed steps to restart. The engine resets those steps and their downstream chain, then re-activates them. Warnings are shown for cross-branch targets.

**Architecture:** New top-level engine method `repeatToSteps()` (not via submitAction — REPEAT doesn't complete a step). Uses the existing `activateStep()` which already handles resetting COMPLETED→IDLE. The key new logic is the forward BFS to collect downstream steps that need resetting. UI adds a Repeat button + modal step picker.

**Tech Stack:** TypeScript, React 19, CSS Modules, Lucide React

```typescript
/** Get all user-visible completed steps (excludes auto-completing types). */
getCompletedSteps(): CompletedStepInfo[] {
  const autoTypes = new Set(['START', 'END', 'PARALLEL', 'WAIT ALL', 'WAIT ANY', 'SELECT 1', 'SELECT_1', 'SCRIPT', 'MATH']);
  const result: CompletedStepInfo[] = [];
  for (const step of this.steps.values()) {
    if (step.state !== 'COMPLETED') continue;
    if (autoTypes.has(canonicalStepType(step.stepType))) continue;
    // Find completion order from trace
    const traceEntry = this.trace.find(t => t.step_oid === step.oid && t.state === 'COMPLETED');
    result.push({
      oid: step.oid,
      localId: step.step.local_id,
      stepType: step.stepType,
      description: step.step.description ?? step.stepType,
      completedOrder: traceEntry?.order ?? 0,
    });
  }
  result.sort((a, b) => a.completedOrder - b.completedOrder);
  return result;
}
```

```typescript
/** Check if repeating to target steps is safe (on the same branch). Returns warning strings. */
checkRepeatSafety(targetOids: string[]): string[] {
  const warnings: string[] = [];

  // Build backward adjacency map: step_oid → set of predecessor step_oids
  const predecessors = new Map<string, Set<string>>();
  for (const conn of this.connections) {
    const fromStep = this.findStepByLocalId(conn.from_step_id);
    const toStep = this.findStepByLocalId(conn.to_step_id);
    if (!fromStep || !toStep) continue;
    if (!predecessors.has(toStep.oid)) predecessors.set(toStep.oid, new Set());
    predecessors.get(toStep.oid)!.add(fromStep.oid);
  }

  // Get all currently active step OIDs
  const activeOids = new Set<string>();
  for (const step of this.steps.values()) {
    if (ACTIVE_STEP_STATES.has(step.state)) activeOids.add(step.oid);
  }

  // For each active step, walk backward to build ancestor set
  const ancestors = new Set<string>();
  const queue = [...activeOids];
  while (queue.length > 0) {
    const current = queue.pop()!;
    const preds = predecessors.get(current);
    if (!preds) continue;
    for (const pred of preds) {
      if (!ancestors.has(pred)) {
        ancestors.add(pred);
        queue.push(pred);
      }
    }
  }

  for (const targetOid of targetOids) {
    if (!ancestors.has(targetOid) && !activeOids.has(targetOid)) {
      const step = this.steps.get(targetOid);
      const desc = step?.step.description ?? step?.stepType ?? targetOid;
      warnings.push(`Step "${desc}" is on a different branch — workflow may not execute correctly`);
    }
  }

  return warnings;
}

private findStepByLocalId(localId: string): StepInstance | undefined {
  for (const step of this.steps.values()) {
    if (step.step.local_id === localId) return step;
  }
  return undefined;
}
```

```typescript
/** Repeat execution from target steps. Resets downstream steps and re-activates targets. */
repeatToSteps(targetOids: string[]): void {
  if (this.workflowState !== 'RUNNING') return;

  // Validate: all targets must be COMPLETED
  for (const oid of targetOids) {
    const step = this.steps.get(oid);
    if (!step || step.state !== 'COMPLETED') {
      throw new Error(`Cannot repeat to step ${oid}: not in COMPLETED state`);
    }
  }

  // Build forward adjacency map: step_oid → set of successor step_oids
  const successors = new Map<string, Set<string>>();
  for (const conn of this.connections) {
    const fromStep = this.findStepByLocalId(conn.from_step_id);
    const toStep = this.findStepByLocalId(conn.to_step_id);
    if (!fromStep || !toStep) continue;
    if (!successors.has(fromStep.oid)) successors.set(fromStep.oid, new Set());
    successors.get(fromStep.oid)!.add(toStep.oid);
  }

  // BFS forward from target steps to collect all downstream steps to reset
  const toReset = new Set<string>(targetOids);
  const bfsQueue = [...targetOids];
  while (bfsQueue.length > 0) {
    const current = bfsQueue.shift()!;
    const succs = successors.get(current);
    if (!succs) continue;
    for (const succ of succs) {
      if (!toReset.has(succ)) {
        toReset.add(succ);
        bfsQueue.push(succ);
      }
    }
  }

  // Reset all collected steps
  for (const oid of toReset) {
    const step = this.steps.get(oid);
    if (!step) continue;
    step.state = 'IDLE';
    this.routingContext.delete(oid);
    this.pendingResources.delete(oid);
    this.pendingUserSteps.delete(oid);
    this.stepParameterSnapshots.delete(oid);
    this.waitAllTracking.delete(oid);
    // If this step had an active child engine, remove it
    if (this.activeChildEngines.has(oid)) {
      this.activeChildEngines.delete(oid);
    }
  }

  // Clear completion queue of any reset steps
  this.completionQueue = this.completionQueue.filter(oid => !toReset.has(oid));

  // Record trace entry for repeat
  for (const oid of targetOids) {
    this.recordTrace(oid, 'EXECUTING');
  }

  // Re-activate target steps
  for (const oid of targetOids) {
    const step = this.steps.get(oid)!;
    this.activateStep(step);
  }

  this.drainCompletionQueue();
}
```

```typescript
      { from_step_id: 's2', to_step_id: 's3' },
      { from_step_id: 's3', to_step_id: 's4' },
    ],
  } as unknown as MasterWorkflowSpecification;

describe('repeatToSteps', () => {
  it('repeats a completed step in a linear chain', () => {
    const engine = new WorkflowEngine(makeLinearWorkflow());
    engine.start();
    // s2 should be executing
    assert.equal(engine.getExecutingSteps().length, 1);
    assert.equal(engine.getExecutingSteps()[0].oid, 's2');

    // Complete s2
    engine.submitAction({ step_oid: 's2', action: 'button_press', button_output: 'go' }, 0);
    assert.equal(engine.getExecutingSteps()[0].oid, 's3');

    // Repeat back to s2
    engine.repeatToSteps(['s2']);
    assert.equal(engine.getWorkflowState(), 'RUNNING');
    const executing = engine.getExecutingSteps();
    assert.equal(executing.length, 1);
    assert.equal(executing[0].oid, 's2');
  });

  it('returns completed steps excluding auto-completing types', () => {
    const engine = new WorkflowEngine(makeLinearWorkflow());
    engine.start();
    engine.submitAction({ step_oid: 's2', action: 'button_press', button_output: 'go' }, 0);
    const completed = engine.getCompletedSteps();
    // Should include s2 (USER_INTERACTION) but not s1 (START)
    assert.equal(completed.length, 1);
    assert.equal(completed[0].oid, 's2');
  });

  it('throws when repeating a non-COMPLETED step', () => {
    const engine = new WorkflowEngine(makeLinearWorkflow());
    engine.start();
    assert.throws(() => engine.repeatToSteps(['s3']), /not in COMPLETED state/);
  });

  it('repeat the same step twice', () => {
    const engine = new WorkflowEngine(makeLinearWorkflow());
    engine.start();
    engine.submitAction({ step_oid: 's2', action: 'button_press', button_output: 'go' }, 0);
    engine.repeatToSteps(['s2']);
    engine.submitAction({ step_oid: 's2', action: 'button_press', button_output: 'go' }, 1);
    engine.repeatToSteps(['s2']);
    assert.equal(engine.getExecutingSteps()[0].oid, 's2');
  });
});
```

```typescript
import { useState, useMemo } from 'react';
import { AlertTriangle } from 'lucide-react';
import type { CompletedStepInfo } from '@engine/types.js';
import styles from './RepeatStepPicker.module.css';

interface RepeatStepPickerProps {
  completedSteps: CompletedStepInfo[];
  checkSafety: (oids: string[]) => string[];
  onConfirm: (targetOids: string[]) => void;
  onCancel: () => void;
}

export function RepeatStepPicker({ completedSteps, checkSafety, onConfirm, onCancel }: RepeatStepPickerProps) {
  const [selected, setSelected] = useState<Set<string>>(new Set());

  const warnings = useMemo(() => {
    if (selected.size === 0) return [];
    return checkSafety([...selected]);
  }, [selected, checkSafety]);

  const stepsWithWarnings = useMemo(() => {
    const warnedOids = new Set<string>();
    for (const step of completedSteps) {
      const stepWarnings = checkSafety([step.oid]);
      if (stepWarnings.length > 0) warnedOids.add(step.oid);
    }
    return warnedOids;
  }, [completedSteps, checkSafety]);

  const toggle = (oid: string) => {
    setSelected(prev => {
      const next = new Set(prev);
      if (next.has(oid)) next.delete(oid);
      else next.add(oid);
      return next;
    });
  };

  return (
    <div className={styles.overlay} onClick={onCancel}>
      <div className={styles.panel} onClick={(e) => e.stopPropagation()}>
        <h3 className={styles.title}>Repeat from Step</h3>

        {warnings.length > 0 && (
          <div className={styles.warningBanner}>
            {warnings.map((w, i) => <div key={i}>{w}</div>)}
          </div>
        )}

        <ul className={styles.list}>
          {completedSteps.map(step => (
            <li key={step.oid} className={styles.item} onClick={() => toggle(step.oid)}>
              <input
                type="checkbox"
                className={styles.itemCheckbox}
                checked={selected.has(step.oid)}
                onChange={() => toggle(step.oid)}
              />
              <div className={styles.itemInfo}>
                <div className={styles.itemType}>{step.stepType}</div>
                <div className={styles.itemDesc}>{step.description}</div>
              </div>
              {stepsWithWarnings.has(step.oid) && (
                <AlertTriangle size={16} className={styles.warningBadge} />
              )}
            </li>
          ))}
        </ul>

        <div className={styles.actions}>
          <button className={styles.cancelButton} onClick={onCancel}>Cancel</button>
          <button
            className={styles.confirmButton}
            disabled={selected.size === 0}
            onClick={() => onConfirm([...selected])}
          >
            Confirm Repeat
          </button>
        </div>
      </div>
    </div>
  );
}
```
